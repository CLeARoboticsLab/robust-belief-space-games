463 # Compute `det` from the number of Householder reflections. Handle
464 # the case `Q.τ` contains zeros.
465 _det_tau(τs::AbstractVector{<:Real}) =
466 isodd(count(!iszero, τs)) ? -one(eltype(τs)) : one(eltype(τs))

468 # In complex case, we need to compute the non-unit eigenvalue `λ = 1 - c*τ`
469 # (where `c = v'v`) of each Householder reflector. As we know that the
470 # reflector must have the determinant of 1, it must satisfy `abs2(λ) == 1`.
471 # Combining this with the constraint `c > 0`, it turns out that the eigenvalue
472 # (hence the determinant) can be computed as `λ = -sign(τ)^2`.
473 # See: https://github.com/JuliaLang/julia/pull/32887#issuecomment-521935716
474 _det_tau(τs) = prod(τ -> iszero(τ) ? one(τ) : -sign(τ)^2, τs)

480 """
481 HessenbergQ <: AbstractQ
482
483 Given a [`Hessenberg`](@ref) factorization object `F`, `F.Q` returns
484 a `HessenbergQ` object, which is an implicit representation of the unitary
485 matrix `Q` in the Hessenberg factorization `QHQ'` represented by `F`.
486 This `F.Q` object can be efficiently multiplied by matrices or vectors,
487 and can be converted to an ordinary matrix type with `Matrix(F.Q)`.
488 """
489 struct HessenbergQ{T,S<:AbstractMatrix,W<:AbstractVector,sym} <: AbstractQ{T}
490 uplo::Char
491 factors::S
492 τ::W
493 function HessenbergQ{T,S,W,sym}(uplo::AbstractChar, factors, τ) where {T,S<:AbstractMatrix,W<:AbstractVector,sym}
494 new(uplo, factors, τ)
495 end
496 end
497 HessenbergQ(F::Hessenberg{<:Any,<:UpperHessenberg,S,W}) where {S,W} = HessenbergQ{eltype(F.factors),S,W,false}(F.uplo, F.factors, F.τ)
498 HessenbergQ(F::Hessenberg{<:Any,<:SymTridiagonal,S,W}) where {S,W} = HessenbergQ{eltype(F.factors),S,W,true}(F.uplo, F.factors, F.τ)
499
500 size(Q::HessenbergQ, dim::Integer) = size(getfield(Q, :factors), dim == 2 ? 1 : dim)
501 size(Q::HessenbergQ) = size(Q, 1), size(Q, 2)
502
503 # HessenbergQ from LAPACK/BLAS (as opposed to Julia libraries like GenericLinearAlgebra)
504 const BlasHessenbergQ{T,sym} = HessenbergQ{T,<:StridedMatrix{T},<:StridedVector{T},sym} where {T<:BlasFloat,sym}
505
506 ## reconstruct the original matrix
507 Matrix{T}(Q::BlasHessenbergQ{<:Any,false}) where {T} = convert(Matrix{T}, LAPACK.orghr!(1, size(Q.factors, 1), copy(Q.factors), Q.τ))
508 Matrix{T}(Q::BlasHessenbergQ{<:Any,true}) where {T} = convert(Matrix{T}, LAPACK.orgtr!(Q.uplo, copy(Q.factors), Q.τ))
509
510 lmul!(Q::BlasHessenbergQ{T,false}, X::StridedVecOrMat{T}) where {T<:BlasFloat} =
511 LAPACK.ormhr!('L', 'N', 1, size(Q.factors, 1), Q.factors, Q.τ, X)
512 rmul!(X::StridedVecOrMat{T}, Q::BlasHessenbergQ{T,false}) where {T<:BlasFloat} =
513 LAPACK.ormhr!('R', 'N', 1, size(Q.factors, 1), Q.factors, Q.τ, X)
514 lmul!(adjQ::AdjointQ{<:Any,<:BlasHessenbergQ{T,false}}, X::StridedVecOrMat{T}) where {T<:BlasFloat} =
515 (Q = adjQ.Q; LAPACK.ormhr!('L', ifelse(T<:Real, 'T', 'C'), 1, size(Q.factors, 1), Q.factors, Q.τ, X))
516 rmul!(X::StridedVecOrMat{T}, adjQ::AdjointQ{<:Any,<:BlasHessenbergQ{T,false}}) where {T<:BlasFloat} =
517 (Q = adjQ.Q; LAPACK.ormhr!('R', ifelse(T<:Real, 'T', 'C'), 1, size(Q.factors, 1), Q.factors, Q.τ, X))
518
519 lmul!(Q::BlasHessenbergQ{T,true}, X::StridedVecOrMat{T}) where {T<:BlasFloat} =
520 LAPACK.ormtr!('L', Q.uplo, 'N', Q.factors, Q.τ, X)
521 rmul!(X::StridedVecOrMat{T}, Q::BlasHessenbergQ{T,true}) where {T<:BlasFloat} =
522 LAPACK.ormtr!('R', Q.uplo, 'N', Q.factors, Q.τ, X)
523 lmul!(adjQ::AdjointQ{<:Any,<:BlasHessenbergQ{T,true}}, X::StridedVecOrMat{T}) where {T<:BlasFloat} =
524 (Q = adjQ.Q; LAPACK.ormtr!('L', Q.uplo, ifelse(T<:Real, 'T', 'C'), Q.factors, Q.τ, X))
525 rmul!(X::StridedVecOrMat{T}, adjQ::AdjointQ{<:Any,<:BlasHessenbergQ{T,true}}) where {T<:BlasFloat} =
526 (Q = adjQ.Q; LAPACK.ormtr!('R', Q.uplo, ifelse(T<:Real, 'T', 'C'), Q.factors, Q.τ, X))
527
528 lmul!(Q::HessenbergQ{T}, X::Adjoint{T,<:StridedVecOrMat{T}}) where {T} = rmul!(X', Q')'
529 rmul!(X::Adjoint{T,<:StridedVecOrMat{T}}, Q::HessenbergQ{T}) where {T} = lmul!(Q', X')'
530 lmul!(adjQ::AdjointQ{<:Any,<:HessenbergQ{T}}, X::Adjoint{T,<:StridedVecOrMat{T}}) where {T} = rmul!(X', adjQ')'
531 rmul!(X::Adjoint{T,<:StridedVecOrMat{T}}, adjQ::AdjointQ{<:Any,<:HessenbergQ{T}}) where {T} = lmul!(adjQ', X')'

533 # flexible left-multiplication (and adjoint right-multiplication)
534 qsize_check(Q::Union{QRPackedQ,QRCompactWYQ,HessenbergQ}, B::AbstractVecOrMat) =
535 size(B, 1) in size(Q.factors) ||
536 throw(DimensionMismatch(lazy"first dimension of B, $(size(B,1)), must equal one of the dimensions of Q, $(size(Q.factors))"))
537 qsize_check(A::AbstractVecOrMat, adjQ::AdjointQ{<:Any,<:Union{QRPackedQ,QRCompactWYQ,HessenbergQ}}) =
538 (Q = adjQ.Q; size(A, 2) in size(Q.factors) ||
539 throw(DimensionMismatch(lazy"second dimension of A, $(size(A,2)), must equal one of the dimensions of Q, $(size(Q.factors))")))

546
547 struct LQPackedQ{T,S<:AbstractMatrix{T},C<:AbstractVector{T}} <: AbstractQ{T}
548 factors::S
549 τ::C
550 end
551
552 LQPackedQ{T}(Q::LQPackedQ) where {T} = LQPackedQ(convert(AbstractMatrix{T}, Q.factors), convert(AbstractVector{T}, Q.τ))
553 @deprecate(AbstractMatrix{T}(Q::LQPackedQ) where {T},
554 convert(AbstractQ{T}, Q),
555 false)
556 Matrix{T}(A::LQPackedQ) where {T} = convert(Matrix{T}, LAPACK.orglq!(copy(A.factors), A.τ))
557 convert(::Type{AbstractQ{T}}, Q::LQPackedQ) where {T} = LQPackedQ{T}(Q)
558
559 # size(Q::LQPackedQ) yields the shape of Q's square form
560 size(Q::LQPackedQ) = (n = size(Q.factors, 2); return n, n)
561
562 ## Multiplication
563 # out-of-place right application of LQPackedQs
564 #
565 # these methods: (1) check whether the applied-to matrix's (A's) appropriate dimension
566 # (columns for A_*, rows for Ac_*) matches the number of columns (nQ) of the LQPackedQ (Q),
567 # and if so effectively apply Q's square form to A without additional shenanigans; and
568 # (2) if the preceding dimensions do not match, check whether the appropriate dimension of
569 # A instead matches the number of rows of the matrix of which Q is a factor (i.e.
570 # size(Q.factors, 1)), and if so implicitly apply Q's truncated form to A by zero extending
571 # A as necessary for check (1) to pass (if possible) and then applying Q's square form
572
573 qsize_check(adjQ::AdjointQ{<:Any,<:LQPackedQ}, B::AbstractVecOrMat) =
574 size(B, 1) in size(adjQ.Q.factors) ||
575 throw(DimensionMismatch(lazy"first dimension of B, $(size(B,1)), must equal one of the dimensions of Q, $(size(adjQ.Q.factors))"))
576 qsize_check(A::AbstractVecOrMat, Q::LQPackedQ) =
577 size(A, 2) in size(Q.factors) ||
578 throw(DimensionMismatch(lazy"second dimension of A, $(size(A,2)), must equal one of the dimensions of Q, $(size(Q.factors))"))
579
580 # in-place right-application of LQPackedQs
581 # these methods require that the applied-to matrix's (A's) number of columns
582 # match the number of columns (nQ) of the LQPackedQ (Q) (necessary for in-place
583 # operation, and the underlying LAPACK routine (ormlq) treats the implicit Q
584 # as its (nQ-by-nQ) square form)
585 rmul!(A::StridedVecOrMat{T}, B::LQPackedQ{T}) where {T<:BlasFloat} =
586 LAPACK.ormlq!('R', 'N', B.factors, B.τ, A)
587 rmul!(A::StridedVecOrMat{T}, adjB::AdjointQ{<:Any,<:LQPackedQ{T}}) where {T<:BlasReal} =
588 (B = adjB.Q; LAPACK.ormlq!('R', 'T', B.factors, B.τ, A))
589 rmul!(A::StridedVecOrMat{T}, adjB::AdjointQ{<:Any,<:LQPackedQ{T}}) where {T<:BlasComplex} =
590 (B = adjB.Q; LAPACK.ormlq!('R', 'C', B.factors, B.τ, A))
591
592 ### QB / QcB
593 lmul!(A::LQPackedQ{T}, B::StridedVecOrMat{T}) where {T<:BlasFloat} = LAPACK.ormlq!('L','N',A.factors,A.τ,B)
594 lmul!(adjA::AdjointQ{<:Any,<:LQPackedQ{T}}, B::StridedVecOrMat{T}) where {T<:BlasReal} =
595 (A = adjA.Q; LAPACK.ormlq!('L', 'T', A.factors, A.τ, B))
596 lmul!(adjA::AdjointQ{<:Any,<:LQPackedQ{T}}, B::StridedVecOrMat{T}) where {T<:BlasComplex} =
597 (A = adjA.Q; LAPACK.ormlq!('L', 'C', A.factors, A.τ, B))
598
599 # In LQ factorization, `Q` is expressed as the product of the adjoint of the
600 # reflectors. Thus, `det` has to be conjugated.
601 det(Q::LQPackedQ) = conj(_det_tau(Q.τ))
