87 function unsafe_load_commands(v::Ptr{Ptr{UInt8}})
88 cmds = Pair{Char, String}[]
89 v == C_NULL && return cmds
90 i = 1
91 while true
92 s = unsafe_load(v, i)
93 s == C_NULL && break
94 e = Char(unsafe_load(s))
95 push!(cmds, e => unsafe_string(s + 1))
96 i += 1
97 end
