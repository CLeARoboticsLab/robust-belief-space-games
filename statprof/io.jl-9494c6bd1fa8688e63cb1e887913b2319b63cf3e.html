239 """
240 write(io::IO, x)
241
242 Write the canonical binary representation of a value to the given I/O stream or file.
243 Return the number of bytes written into the stream. See also [`print`](@ref) to
244 write a text representation (with an encoding that may depend upon `io`).
245
246 The endianness of the written value depends on the endianness of the host system.
247 Convert to/from a fixed endianness when writing/reading (e.g. using [`htol`](@ref) and
248 [`ltoh`](@ref)) to get results that are consistent across platforms.
249
250 You can write multiple values with the same `write` call. i.e. the following are equivalent:
251
252 write(io, x, y...)
253 write(io, x) + write(io, y...)

255 # Examples
256 Consistent serialization:
257 ```jldoctest
258 julia> fname = tempname(); # random temporary filename
259
260 julia> open(fname,"w") do f
261 # Make sure we write 64bit integer in little-endian byte order
262 write(f,htol(Int64(42)))
263 end
264 8
265
266 julia> open(fname,"r") do f
267 # Convert back to host byte order and host integer type
268 Int(ltoh(read(f,Int64)))
269 end
270 42
271 ```
272
273 Merging write calls:
274 ```jldoctest
275 julia> io = IOBuffer();
276
277 julia> write(io, "JuliaLang is a GitHub organization.", " It has many members.")

279
280 julia> String(take!(io))
281 "JuliaLang is a GitHub organization. It has many members."
282
283 julia> write(io, "Sometimes those members") + write(io, " write documentation.")

353 """
354 open_flags(; keywords...) -> NamedTuple
355
356 Compute the `read`, `write`, `create`, `truncate`, `append` flag value for
357 a given set of keyword arguments to [`open`](@ref) a [`NamedTuple`](@ref).
358 """
359 function open_flags(;
360 read :: Union{Bool,Nothing} = nothing,
361 write :: Union{Bool,Nothing} = nothing,
362 create :: Union{Bool,Nothing} = nothing,
363 truncate :: Union{Bool,Nothing} = nothing,
364 append :: Union{Bool,Nothing} = nothing,
365 )
366 if write === true && read !== true && append !== true
367 create === nothing && (create = true)
368 truncate === nothing && (truncate = true)
369 end
370
371 if truncate === true || append === true
372 write === nothing && (write = true)
373 create === nothing && (create = true)
374 end
375
376 write === nothing && (write = false)
377 read === nothing && (read = !write)
378 create === nothing && (create = false)
379 truncate === nothing && (truncate = false)
380 append === nothing && (append = false)
381
382 return (
383 read = read,
384 write = write,
385 create = create,
386 truncate = truncate,
387 append = append,
388 )
389 end

657 # fallback to optimized methods for IOBuffer in iobuffer.jl
658 function copyline(out::IO, s::IO; keep::Bool=false)
659 if keep
660 return copyuntil(out, s, 0x0a, keep=true)
661 else
662 # more complicated to deal with CRLF logic
663 while !eof(s)
664 b = read(s, UInt8)
665 b == 0x0a && break
666 if b == 0x0d && !eof(s)
667 b = read(s, UInt8)
668 b == 0x0a && break
669 write(out, 0x0d)
670 end
671 write(out, b)
672 end
673 return out
674 end

1010 # requires that indices for target are the integer unit range from firstindex to lastindex
1011 # returns whether the delimiter was matched
1012 # uses the Knuth–Morris–Pratt_algorithm, with the first and second cache entries unrolled
1013 # For longer targets, the cache improves the big-O efficiency of scanning of sequences
1014 # with repeated patterns
1015 # Each cache entry tells us which index we should start the search at.
1016 # We assume this is unlikely, so we only lazy-initialize as much of the cache as we need to use
1017 # When we allocate the cache, we initialize it to 0 (and offset by the first index afterwards).
1018 # Suppose target is:
1019 # Index: 1245689
1020 # Value: "aδcaδcx"
1021 # We would set the cache to
1022 # 0 0 0 1 2 3 4 0
1023 # So after if we mismatch after the second aδc sequence,
1024 # we can immediately jump back to index 5 (4 + 1).
1025 function readuntil_vector!(io::IO, target::AbstractVector{T}, keep::Bool, out) where {T}
1026 first = firstindex(target)
1027 last = lastindex(target)
1028 len = last - first + 1
1029 if len < 1
1030 return true
1031 end
1032 pos = 0 # array-offset
1033 max_pos = 1 # array-offset in cache
1034 local cache # will be lazy initialized when needed
1035 output! = (isa(out, IO) ? write : push!)
1036 for c in readeach(io, T)
1037 # Backtrack until the next target character matches what was found
1038 while true
1039 c1 = target[pos + first]
1040 if c == c1
1041 pos += 1
1042 break
1043 elseif pos == 0
1044 break
1045 elseif pos == 1
1046 if !keep
1047 output!(out, target[first])
1048 end
1049 pos = 0
1050 else
1051 # grow cache to contain up to `pos`
1052 if !@isdefined(cache)
1053 cache = zeros(Int, len)
1054 end
1055 while max_pos < pos
1056 ci = target[max_pos + first]
1057 b = max_pos
1058 max_pos += 1
1059 while b != 0
1060 b = cache[b]
1061 cb = target[b + first]
1062 if ci == cb
1063 cache[max_pos] = b + 1
1064 break
1065 end
1066 end
1067 end
1068 # read new position from cache
1069 pos1 = cache[pos]
1070 if !keep
1071 # and add the removed prefix from the target to the output
1072 # if not always keeping the match
1073 for b in 1:(pos - pos1)
1074 output!(out, target[b - 1 + first])
1075 end
1076 end
1077 pos = pos1
1078 end
1079 end
1080 if keep || pos == 0
1081 output!(out, c)
1082 end
1083 pos == len && return true
1084 end
1085 if !keep
1086 # failed early without finishing the match,
1087 # add the partial match to the output
1088 # if not always keeping the match
1089 for b in 1:pos
1090 output!(out, target[b - 1 + first])
1091 end
1092 end
1093 return false
1094 end

1314 while i != ichunk
1315 chunk = chunks[i]
1316 p -= length(resize!(chunk, min(4096, p-p0)))
1317 readbytes!(seek(r.itr.stream, p), chunk)
1318 i = i == 1 ? length(chunks) : i - 1
1319 end
1320 end
1321
1322 # find the newline previous to inewline
1323 jchunk = ichunk
1324 jnewline = inewline
1325 while true
1326 inewline = something(findprev(==(UInt8('\n')), chunks[ichunk], inewline-1), 0)
1327 inewline > 0 && break
1328 ichunk = ichunk == 1 ? length(chunks) : ichunk - 1
1329 ichunk == jchunk && break # found nothing — may need to read more chunks
1330 inewline = length(chunks[ichunk])+1 # start for next findprev
1331 end
1332
1333 # read more chunks to look for a newline (should rarely happen)
1334 if inewline == 0 && p > p0
1335 ichunk = jchunk + 1
1336 while true
1337 chunk = Vector{UInt8}(undef, min(4096, p-p0))
1338 p -= length(chunk)
1339 readbytes!(seek(r.itr.stream, p), chunk)
1340 insert!(chunks, ichunk, chunk)
1341 inewline = something(findlast(==(UInt8('\n')), chunk), 0)
1342 (p == p0 || inewline > 0) && break
1343 end
1344 end
1345 end
1346 return (s, (; p0, p, chunks, ichunk, inewline, jchunk, jnewline))
1347 end
1348 isdone(r::Iterators.Reverse{<:EachLine}, state) = isempty(state.chunks)
1349 isdone(r::Iterators.Reverse{<:EachLine}) = isdone(r.itr)
1350
1351 # use reverse iteration to get end of EachLines (if possible)
1352 last(itr::EachLine) = first(Iterators.reverse(itr))

1449 """
1450 skipchars(predicate, io::IO; linecomment=nothing)
1451
1452 Advance the stream `io` such that the next-read character will be the first remaining for
1453 which `predicate` returns `false`. If the keyword argument `linecomment` is specified, all
1454 characters from that character until the start of the next line are ignored.
1455
1456 # Examples
1457 ```jldoctest
1458 julia> buf = IOBuffer(" text")
1459 IOBuffer(data=UInt8[...], readable=true, writable=false, seekable=true, append=false, size=8, maxsize=Inf, ptr=1, mark=-1)
1460
1461 julia> skipchars(isspace, buf)
1462 IOBuffer(data=UInt8[...], readable=true, writable=false, seekable=true, append=false, size=8, maxsize=Inf, ptr=5, mark=-1)
1463
1464 julia> String(readavailable(buf))
1465 "text"
1466 ```
1467 """
1468 function skipchars(predicate, io::IO; linecomment=nothing)
1469 for c in readeach(io, Char)
1470 if c === linecomment
1471 readline(io)
1472 elseif !predicate(c)
1473 skip(io, -ncodeunits(c))
1474 break
1475 end
1476 end
1477 return io
1478 end

1480 """
1481 countlines(io::IO; eol::AbstractChar = '\\n')
1482 countlines(filename::AbstractString; eol::AbstractChar = '\\n')
1483
1484 Read `io` until the end of the stream/file and count the number of lines. To specify a file
1485 pass the filename as the first argument. EOL markers other than `'\\n'` are supported by
1486 passing them as the second argument. The last non-empty line of `io` is counted even if it does not
1487 end with the EOL, matching the length returned by [`eachline`](@ref) and [`readlines`](@ref).
1488
1489 To count lines of a `String`, `countlines(IOBuffer(str))` can be used.
1490
1491 # Examples
1492 ```jldoctest
1493 julia> io = IOBuffer("JuliaLang is a GitHub organization.\\n");
1494
1495 julia> countlines(io)
1496 1
1497
1498 julia> io = IOBuffer("JuliaLang is a GitHub organization.");
1499
1500 julia> countlines(io)
1501 1
1502
1503 julia> eof(io) # counting lines moves the file pointer
1504 true
1505
1506 julia> io = IOBuffer("JuliaLang is a GitHub organization.");
1507
1508 julia> countlines(io, eol = '.')
1509 1
1510 ```
1511 ```jldoctest
1512 julia> write("my_file.txt", "JuliaLang is a GitHub organization.\\n")
1513 36
1514
1515 julia> countlines("my_file.txt")
1516 1
1517
1518 julia> countlines("my_file.txt", eol = 'n')
1519 4
1520
1521 julia> rm("my_file.txt")
1522
1523 ```
1524 """
1525 function countlines(io::IO; eol::AbstractChar='\n')
1526 isascii(eol) || throw(ArgumentError("only ASCII line terminators are supported"))
1527 aeol = UInt8(eol)
1528 a = Vector{UInt8}(undef, 8192)
1529 nl = nb = 0
1530 while !eof(io)
1531 nb = readbytes!(io, a)
1532 @simd for i=1:nb
1533 @inbounds nl += a[i] == aeol
1534 end
1535 end
1536 if nb > 0 && a[nb] != aeol
1537 nl += 1 # final line is not terminated with eol
1538 end
1539 nl
1540 end
1541
1542 countlines(f::AbstractString; eol::AbstractChar = '\n') = open(io->countlines(io, eol = eol), f)::Int
